        if st.session_state.monitoring and sel:
            base = base_map[sel]
            curve = json.loads(base['curve_data'])
            tol = base['tolerance_pct']
            
            idx = st.session_state.monitor_index
            
            if idx < len(curve):
                # 處理當前數據點
                golden = curve[idx]
                real = golden + np.random.normal(0, 5)
                
                # Demo 模式異常注入
                if os.getenv('DEMO_MODE', 'false').lower() == 'true':
                    if 50 < idx < 70:
                        real += golden * 0.4  # 注入 40% 偏差
                
                st.session_state.rt_data.append(real)
                var_pct = abs(real - golden) / (abs(golden) + 0.01) * 100
                
                # ========== 紅綠燈判定 ==========
                if var_pct <= tol:
                    status, color = T["st_g"], "normal"
                    st.session_state.in_alarm = False  # 恢復正常
                elif var_pct <= tol * 1.5:
                    status, color = T["st_y"], "off"
                    # 黃燈不改變 alarm 狀態
                else:
                    status, color = T["st_r"], "inverse"
                    
                    # ========== 自動發單（防重複）==========
                    if not st.session_state.in_alarm:
                        try:
                            conn = get_conn()
                            if conn:
                                cur = conn.cursor()
                                cur.execute(
                                    "INSERT INTO asns_correction_tasks "
                                    "(baseline_id, suid, cuid, max_variance, status) "
                                    "VALUES (%s, %s, %s, %s, 0)",
                                    (base['baseline_id'], base['suid'], base['cuid'], var_pct)
                                )
                                conn.commit()
                                conn.close()
                                st.toast(f"🚨 {T['st_r']}", icon="🔴")
                        except Exception as e:
                            st.error(f"發單失敗：{e}")
                        
                        st.session_state.in_alarm = True  # 標記已發單
                
                # ========== 更新 UI ==========
                with ph_metric.container():
                    m1, m2, m3 = st.columns(3)
                    m1.metric(T["curr"], f"{real:.2f}")
                    m2.metric(T["dev"], f"{var_pct:.1f}%", delta_color=color)
                    
                    if color == "inverse":
                        m3.error(status)
                    elif color == "off":
                        m3.warning(status)
                    else:
                        m3.success(status)
                
                with ph_chart.container():
                    # 準備圖表數據
                    chart_data = pd.DataFrame({
                        "Golden": curve[:idx+1],
                        "Real-time": st.session_state.rt_data
                    })
                    st.line_chart(
                        chart_data,
                        color=["#CCCCCC", "#FF4B4B"],
                        height=400
                    )
                
                # ========== 推進索引並重新運行 ==========
                st.session_state.monitor_index += 1
                time.sleep(0.1)
                st.rerun()
            else:
                # 監控完成
                st.session_state.monitoring = False
                st.info("✅ 監控完成")
        
        elif not st.session_state.monitoring and len(st.session_state.rt_data) > 0:
            # 顯示最後狀態
            with ph_chart.container():

                    "Golden": curve,
                    "Real-time": st.session_state.rt_data + [None] * (len(curve) - len(st.session_state.rt_data))
                })
                st.line_chart(
                    chart_data,
                    color=["#CCCCCC", "#FF4B4B"],
                    height=400
                )

def generate_curve():
    """生成模擬黃金曲線"""
    x = np.linspace(0, 12, 200)
    y = 100 * (1 - np.exp(-0.5 * x)) + np.random.normal(0, 1, 200)
    return y.tolist()

c1, c2 = st.columns([3, 1])

with c1:
    st.caption("預覽曲線")
    preview_curve = generate_curve()
    st.line_chart(preview_curve, height=400)

with c2:
    st.caption("基線設定")
    n = st.text_input("Name", "Golden_01")
    suid = st.text_input("SUID", "S1")
    cuid = st.text_input("CUID", "C1")
    t = st.slider("Tolerance %", 1, 20, 10)
    
    if st.button(T["btn_save"], type="primary", use_container_width=True):
        try:
            conn = get_conn()
            if conn:
                cur = conn.cursor()
                cur.execute(
                    "INSERT INTO asns_golden_baselines "
                    "(baseline_name, suid, cuid, tolerance_pct, curve_data) "
                    "VALUES (%s, %s, %s, %s, %s)",
                    (n, suid, cuid, t, json.dumps(generate_curve()))
                )
                conn.commit()
                conn.close()
                st.success(T["msg_saved"])
                time.sleep(1)
                st.rerun()
        except Exception as e:
            st.error(f"儲存失敗：{e}")
